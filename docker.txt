# 1. Build the images
docker-compose build

# 2. Start the containers
docker-compose up -d

# 3. Check status
docker-compose ps

# 4. View logs
docker-compose logs -f

Stop containers:  docker-compose down

Restart containers:  docker-compose restart

View container status:  docker-compose ps

# 1. Pull images from Docker Hub
docker pull suhasvr/task_manager_agent:backend-latest
docker pull suhasvr/task_manager_agent:frontend-latest

# 2. Start with docker-compose
docker-compose up -d

# Using the script (recommended)
.\build-and-push.ps1 -Tag "v1.0.0"

# Or manually:
docker build -f Dockerfile.backend -t suhasvr/task_manager_agent:backend-v1.0.0 .
docker build -f Dockerfile.frontend -t suhasvr/task_manager_agent:frontend-v1.0.0 .
docker push suhasvr/task_manager_agent:backend-v1.0.0
docker push suhasvr/task_manager_agent:frontend-v1.0.0
